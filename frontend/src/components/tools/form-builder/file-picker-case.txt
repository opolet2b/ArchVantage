            case "file_picker":
                return (
                    <Input
                        type="file"
                        required={widget.required}
                        onChange={(e) => {
                            const file = e.target.files?.[0]
                            if (file) {
                                handleValueChange(widget.id, file.name)
                            }
                        }}
                        disabled={readOnly}
                    />
                )
